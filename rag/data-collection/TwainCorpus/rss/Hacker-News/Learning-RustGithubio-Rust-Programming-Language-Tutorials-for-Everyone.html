fn main(){letmarks=["10","20","30","0.4","0.5","invalid"];letsum: i32 =marks// Iterate through marks
.iter()// Parse string to float, ignore if invalid
.filter_map(|&s|s.parse::<f64>().ok())// Multiply decimals under 1.0 by 100
.map(|num|ifnum<1.0{num*100.0}else{num})// Trace the values
.inspect(|&val|println!("Parsed: {val}"))// Cast float to integer
.map(|num|numasi32)// Aggregate
.sum();println!("Final: {sum}");// 150
}
